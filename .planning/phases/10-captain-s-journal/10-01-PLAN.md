```typescript
/**
 * Insert a journal entry. Call INSIDE an existing transaction for atomicity
 * with the action that triggered the entry (buy, sell, event resolve, state transition).
 * Can also be called standalone for player notes.
 */
export function insertJournalEntry(
  db: Database.Database,
  turn: number,
  entryType: JournalEntryType,
  title: string,
  bodyJson: Record<string, unknown>
): void {
  db.prepare(`
    INSERT INTO journal_entries (turn, entry_type, title, body_json, created_at)
    VALUES (?, ?, ?, ?, datetime('now'))
  `).run(turn, entryType, title, JSON.stringify(bodyJson));
}

/**
 * Retrieve journal entries, newest first.
 * Optional limit parameter (default: all entries).
 */
export function getJournalEntries(db: Database.Database, limit?: number): JournalEntry[] {
  const sql = limit
    ? 'SELECT * FROM journal_entries ORDER BY turn DESC, id DESC LIMIT ?'
    : 'SELECT * FROM journal_entries ORDER BY turn DESC, id DESC';
  const rows = limit
    ? db.prepare(sql).all(limit) as Array<{ id: number; turn: number; entry_type: string; title: string; body_json: string; created_at: string }>
    : db.prepare(sql).all() as Array<{ id: number; turn: number; entry_type: string; title: string; body_json: string; created_at: string }>;
  return rows.map(r => ({
    id: r.id,
    turn: r.turn,
    entryType: r.entry_type as JournalEntryType,
    title: r.title,
    body: JSON.parse(r.body_json),
    createdAt: r.created_at,
  }));
}

/**
 * Delete all journal entries — called on New Game reset.
 */
export function clearJournalEntries(db: Database.Database): void {
  db.prepare('DELETE FROM journal_entries').run();
}
```

```typescript
/**
 * IPC handlers for journal channels.
 *
 * registerJournalIPCHandlers(db) wires:
 *   journal:getEntries — return all persisted journal entries (newest first)
 *   journal:addNote    — add a player-written note to the journal
 */

import type Database from 'better-sqlite3';
import { wrapHandler } from '../lib/ipc-wrapper';
import { createLogger } from '../lib/logger';
import { loadGameState, insertJournalEntry, getJournalEntries } from '../db/queries';
import type { JournalEntry } from '../../shared/game-types';

const log = createLogger('JournalIPC');

export function registerJournalIPCHandlers(db: Database.Database): void {
  /**
   * journal:getEntries — return persisted journal entries, newest first.
   * Optional limit parameter.
   */
  wrapHandler<JournalEntry[], [number?]>(
    'journal:getEntries',
    async (limit) => {
      return getJournalEntries(db, limit);
    },
  );

  /**
   * journal:addNote — player writes a personal note/reflection.
   * Creates a 'note' type journal entry at the current turn.
   */
  wrapHandler<void, [string, string]>(
    'journal:addNote',
    async (title, text) => {
      const state = loadGameState(db);
      const turn = state?.currentTurn ?? 0;
      insertJournalEntry(db, turn, 'note', title, { text });
      log.info('Player note added', { turn, title });
    },
    (title, text) => {
      if (!title || typeof title !== 'string') return 'title required';
      if (!text || typeof text !== 'string') return 'text required';
      return null;
    },
  );
}
```
