**3. Interleave ephemeral weather/rumor entries at render time:**
Create a merged display list that combines DB entries with deterministic weather/rumor entries. Weather and rumor are NOT stored in DB — they are regenerated from the turn numbers present in DB entries plus the current turn:

```typescript
// Build display entries: DB entries + interleaved weather/rumor
const displayEntries = React.useMemo(() => {
  // Collect all turns that have DB entries, plus current turn
  const turns = new Set(dbEntries.map(e => e.turn));
  turns.add(currentTurn);

  type DisplayEntry = {
    id: string;
    turn: number;
    date: string;
    text: string;
    type: LogEntry['type'];
  };

  const all: DisplayEntry[] = [];

  // Add DB entries
  for (const e of dbEntries) {
    all.push({
      id: `db-${e.id}`,
      turn: e.turn,
      date: turnToDate(e.turn),
      text: e.title,
      type: e.entryType === 'voyage_event' ? 'event' : e.entryType as LogEntry['type'],
    });
  }

  // Add weather + rumor for each unique turn
  for (const turn of turns) {
    all.push({
      id: `weather-${turn}`,
      turn,
      date: turnToDate(turn),
      text: getSeasonalWeatherDescription(turn),
      type: 'weather',
    });
    const rumor = generateRumor(turn);
    if (rumor) {
      all.push({
        id: `rumor-${turn}`,
        turn,
        date: turnToDate(turn),
        text: rumor,
        type: 'rumor',
      });
    }
  }

  // Sort newest first (by turn desc, then weather/rumor before DB entries for same turn)
  all.sort((a, b) => {
    if (b.turn !== a.turn) return b.turn - a.turn;
    // Within same turn: weather first, then rumor, then DB entries by id
    const typeOrder = { weather: 0, rumor: 1, arrival: 2, departure: 3, trade: 4, event: 5 };
    return (typeOrder[a.type] ?? 9) - (typeOrder[b.type] ?? 9);
  });

  return all;
}, [dbEntries, currentTurn]);
```
