    ```typescript
    import { useState, useEffect } from 'react';
    import type { Port } from '../../shared/game-types';

    export function useMapPorts() {
      const [ports, setPorts] = useState<Port[]>([]);
      const [loading, setLoading] = useState(true);
      const [error, setError] = useState<string | null>(null);

      useEffect(() => {
        window.api.game.getPorts()
          .then(p => { setPorts(p); setLoading(false); })
          .catch(e => { setError(String(e)); setLoading(false); });
      }, []);

      return { ports, loading, error };
    }
    ```

    **src/renderer/components/PortInfoPanel.tsx** — Create the left-panel content component:
    ```typescript
    import React, { useState, useEffect } from 'react';
    import type { Port, VoyageRoute } from '../../shared/game-types';

    interface ReachableDestination {
      port: Port;
      route: VoyageRoute;
    }

    interface PortInfoPanelProps {
      portId: string;
      currentPortId: string | null;  // player's current location (from gameState)
      onNavigate: (destinationPortId: string, route: VoyageRoute) => void;
    }

    export default function PortInfoPanel({ portId, currentPortId, onNavigate }: PortInfoPanelProps) {
      const [port, setPort] = useState<Port | null>(null);
      const [reachable, setReachable] = useState<ReachableDestination[]>([]);
      const [loading, setLoading] = useState(true);
      // AbortController pattern: ignore stale responses if portId changes rapidly
      const [requestPortId, setRequestPortId] = useState<string | null>(null);

      useEffect(() => {
        setLoading(true);
        setRequestPortId(portId);

        Promise.all([
          window.api.game.getPortDetails(portId),
          window.api.game.getReachablePorts(portId),
        ]).then(([portData, reachableData]) => {
          // Discard if portId changed while this request was in flight
          if (portId !== portId) return; // using closure
          setPort(portData);
          setReachable(reachableData.slice(0, 5)); // show top 5 nearest
          setLoading(false);
        });
      }, [portId]);

      if (loading) {
        return (
          <div className="p-4" style={{ fontFamily: 'Lora, serif', color: '#7a5c45' }}>
            Loading port…
          </div>
        );
      }

      if (!port) return null;

      const isPlayerHere = currentPortId === portId;

      return (
        <div className="p-4 flex flex-col gap-4" style={{ fontFamily: 'Lora, serif' }}>
          {/* Port header */}
          <div>
            <h2 style={{ fontFamily: 'Cinzel, serif', fontSize: '1.1rem', fontWeight: 700, color: '#3d2b1f' }}>
              {port.ancientName}
            </h2>
            <p style={{ fontSize: '0.8rem', color: '#b8894a', fontStyle: 'italic' }}>
              ({port.modernName})
            </p>
            <p style={{ fontSize: '0.85rem', color: '#5a3e2b', marginTop: '0.5rem', lineHeight: 1.5 }}>
              {port.description}
            </p>
          </div>

          {/* Reachable destinations */}
          {isPlayerHere && reachable.length > 0 && (
            <div>
              <h3 style={{ fontFamily: 'Cinzel, serif', fontSize: '0.85rem', fontWeight: 700, color: '#3d2b1f', marginBottom: '0.5rem', textTransform: 'uppercase', letterSpacing: '0.08em' }}>
                Destinations
              </h3>
              <ul className="flex flex-col gap-2">
                {reachable.map(({ port: dest, route }) => (
                  <li key={dest.id} style={{ border: '1px solid #c4956a', borderRadius: 4, padding: '0.5rem', background: 'rgba(240,217,181,0.3)' }}>
                    <div style={{ display: 'flex', justifyContent: 'space-between', alignItems: 'flex-start' }}>
                      <div>
                        <div style={{ fontWeight: 600, fontSize: '0.85rem', color: '#3d2b1f' }}>{dest.ancientName}</div>
                        <div style={{ fontSize: '0.75rem', color: '#b8894a' }}>{Math.round(route.distanceKm)} km · {route.durationTurns} turn{route.durationTurns !== 1 ? 's' : ''}</div>
                      </div>
                      <button
                        onClick={() => onNavigate(dest.id, route)}
                        style={{ fontFamily: 'Cinzel, serif', fontSize: '0.7rem', padding: '3px 8px', background: '#7a5c45', color: '#f0d9b5', border: 'none', borderRadius: 3, cursor: 'pointer' }}
                      >
                        Sail
                      </button>
                    </div>
                  </li>
                ))}
              </ul>
            </div>
          )}

          {!isPlayerHere && (
            <p style={{ fontSize: '0.8rem', color: '#b8894a', fontStyle: 'italic' }}>
              Sail here from your current port to trade and explore.
            </p>
          )}
        </div>
      );
    }
    ```

    **Note on AbortController pattern:** The effect above uses the closure correctly — `portId` in the then() callback is the same as the outer `portId` because it's captured in the closure. If you need stricter staleness detection, use a ref: `const latestPortId = useRef(portId); latestPortId.current = portId;` and check `if (latestPortId.current !== thisPortId) return;` in the then() handler.

    **src/renderer/components/ShipMarker.tsx** — Create the animated ship using Leaflet.motion:
    First install: `npm install leaflet.motion`

    ```typescript
    import React, { useEffect, useRef } from 'react';
    import { useMap } from 'react-leaflet';
    import L from 'leaflet';

    interface ShipMarkerProps {
      fromCoord: [number, number];
      toCoord: [number, number];
      durationMs: number;
      onAnimationEnd?: () => void;
    }

    export default function ShipMarker({ fromCoord, toCoord, durationMs, onAnimationEnd }: ShipMarkerProps) {
      const map = useMap(); // Must be inside MapContainer — this component is rendered as a child
      const motionLayerRef = useRef<any>(null);

      useEffect(() => {
        // Dynamically import leaflet.motion to avoid SSR issues
        import('leaflet.motion/dist/leaflet.motion.js').then(() => {
          // Remove previous animation if re-triggered
          if (motionLayerRef.current) {
            motionLayerRef.current.remove();
          }

          const shipIcon = L.divIcon({
            className: 'ship-icon',
            html: '⛵',
            iconSize: [20, 20],
            iconAnchor: [10, 10],
          });

          // leaflet.motion adds L.motion.polyline to the Leaflet namespace
          const motionLine = (L as any).motion.polyline(
            [fromCoord, toCoord],
            { color: 'rgba(122, 92, 69, 0.4)', weight: 2, dashArray: '4, 4' },
            { auto: true, duration: durationMs, easing: (L as any).Motion.Ease.easeInOutQuad },
            { removeOnEnd: false, showMarker: true, icon: shipIcon }
          );

          motionLine.on('completed', () => {
            if (onAnimationEnd) onAnimationEnd();
          });

          motionLine.addTo(map);
          motionLayerRef.current = motionLine;
        });

        return () => {
          if (motionLayerRef.current) {
            motionLayerRef.current.remove();
            motionLayerRef.current = null;
          }
        };
      }, [fromCoord, toCoord, durationMs, map, onAnimationEnd]);

      return null; // Leaflet manages the DOM
    }
    ```
