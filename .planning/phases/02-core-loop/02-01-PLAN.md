    **src/game/voyage.ts** — Create this file:
    ```typescript
    import type { Port, VoyageRoute } from '../shared/game-types';

    // Haversine formula for great-circle distance between two lat/lng points
    export function haversineDistanceKm(from: Port, to: Port): number {
      const R = 6371;
      const dLat = (to.lat - from.lat) * (Math.PI / 180);
      const dLng = (to.lng - from.lng) * (Math.PI / 180);
      const a =
        Math.sin(dLat / 2) ** 2 +
        Math.cos(from.lat * (Math.PI / 180)) *
          Math.cos(to.lat * (Math.PI / 180)) *
          Math.sin(dLng / 2) ** 2;
      return R * 2 * Math.atan2(Math.sqrt(a), Math.sqrt(1 - a));
    }

    // One turn = one day; ancient merchant ships ~4 knots average (day+night)
    export function calculateVoyageDuration(from: Port, to: Port): number {
      const km = haversineDistanceKm(from, to);
      const nauticalMiles = km / 1.852;
      const days = nauticalMiles / (4 * 24); // 4 knots, 24hrs/day
      return Math.max(1, Math.ceil(days));
    }

    // Returns ports within maxTurns sailing distance (default 5)
    export function getReachablePorts(
      fromPortId: string,
      allPorts: Port[],
      maxTurns: number = 5
    ): Array<{ port: Port; route: VoyageRoute }> {
      const from = allPorts.find(p => p.id === fromPortId);
      if (!from) return [];
      return allPorts
        .filter(p => p.id !== fromPortId)
        .map(to => {
          const distanceKm = haversineDistanceKm(from, to);
          const durationTurns = calculateVoyageDuration(from, to);
          return { port: to, route: { fromPortId, toPortId: to.id, distanceKm, durationTurns } };
        })
        .filter(r => r.route.durationTurns <= maxTurns)
        .sort((a, b) => a.route.durationTurns - b.route.durationTurns);
    }
    ```

    **src/game/economy.ts** — Create this file:

    ```typescript
    import type { MarketGood } from '../shared/game-types';

    // Base price × (1 + (demand - supply) × elasticity), capped ±50%
    export function calculateMarketPrice(
      basePrice: number,
      supply: number,
      demand: number,
      elasticity: number = 0.02
    ): number {
      const delta = demand - supply;
      const multiplier = 1 + delta * elasticity;
      return Math.round(
        Math.max(basePrice * 0.5, Math.min(basePrice * 1.5, basePrice * multiplier))
      );
    }

    // Advance one port's market by one turn: supply ticks add to stock, demand ticks remove
    // Prices are recalculated from updated supply/demand each call
    export function advancePortEconomy(goods: MarketGood[]): MarketGood[] {
      return goods.map(g => ({
        ...g,
        currentPrice: calculateMarketPrice(g.basePrice, g.supply, g.demand),
      }));
    }
    ```
