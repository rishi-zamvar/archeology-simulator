    // Buy a good: deducts gold, adds to cargo, records transaction, updates port supply
    export function buyGood(
      db: Database.Database,
      portId: string,
      goodId: string,
      quantity: number
    ): { success: boolean; error?: string; inventory?: PlayerInventory } {
      const result = db.transaction(() => {
        // Load current inventory
        const inv = getPlayerInventory(db);

        // Load market price for this good at this port
        const market = db.prepare(
          `SELECT pi.supply, pi.demand, g.base_price
           FROM port_inventory pi
           JOIN goods g ON g.id = pi.good_id
           WHERE pi.port_id = ? AND pi.good_id = ?`
        ).get(portId, goodId) as { supply: number; demand: number; base_price: number } | undefined;

        if (!market) throw new Error(`Good ${goodId} not available at ${portId}`);

        // Import calculateMarketPrice here (in main process, use require since CJS)
        // To avoid circular deps: inline the formula
        const delta = market.demand - market.supply;
        const multiplier = 1 + delta * 0.02;
        const price = Math.round(Math.max(market.base_price * 0.5, Math.min(market.base_price * 1.5, market.base_price * multiplier)));
        const totalCost = price * quantity;

        if (inv.gold < totalCost) throw new Error('Insufficient gold');
        if (inv.currentCargo + quantity > inv.cargoCapacity) throw new Error('Insufficient cargo space');

        // Update gold and cargo
        const existingItem = inv.cargoItems.find(c => c.goodId === goodId);
        let updatedCargo: CargoItem[];
        if (existingItem) {
          // Weighted average price for existing stack
          const avgPrice = Math.round(
            (existingItem.boughtAtPrice * existingItem.quantity + price * quantity) /
            (existingItem.quantity + quantity)
          );
          updatedCargo = inv.cargoItems.map(c =>
            c.goodId === goodId
              ? { ...c, quantity: c.quantity + quantity, boughtAtPrice: avgPrice }
              : c
          );
        } else {
          updatedCargo = [...inv.cargoItems, { goodId, quantity, boughtAtPrice: price, boughtAtPort: portId }];
        }

        const newInventory: PlayerInventory = {
          ...inv,
          gold: inv.gold - totalCost,
          cargoItems: updatedCargo,
          currentCargo: inv.currentCargo + quantity,
        };

        // Persist inventory
        db.prepare(
          `UPDATE player_inventory SET gold = ?, cargo_json = ? WHERE id = 1`
        ).run(newInventory.gold, JSON.stringify(newInventory.cargoItems));

        // Record transaction
        const currentState = loadGameState(db);
        db.prepare(
          `INSERT INTO transaction_history (turn, port_id, good_id, action, quantity, price_per_unit, total_value)
           VALUES (?, ?, ?, 'buy', ?, ?, ?)`
        ).run(currentState?.currentTurn ?? 0, portId, goodId, quantity, price, totalCost);

        // Port supply decreases slightly (bought from local stock)
        db.prepare(
          `UPDATE port_inventory SET supply = MAX(0, supply - ?) WHERE port_id = ? AND good_id = ?`
        ).run(Math.floor(quantity / 2), portId, goodId);

        return newInventory;
      })();

      return { success: true, inventory: result };
    }

    // Sell a good: adds gold, removes from cargo, records transaction, updates port supply
    export function sellGood(
      db: Database.Database,
      portId: string,
      goodId: string,
      quantity: number
    ): { success: boolean; error?: string; inventory?: PlayerInventory } {
      const result = db.transaction(() => {
        const inv = getPlayerInventory(db);
        const cargoItem = inv.cargoItems.find(c => c.goodId === goodId);
        if (!cargoItem || cargoItem.quantity < quantity) throw new Error('Insufficient cargo');

        const market = db.prepare(
          `SELECT pi.supply, pi.demand, g.base_price
           FROM port_inventory pi
           JOIN goods g ON g.id = pi.good_id
           WHERE pi.port_id = ? AND pi.good_id = ?`
        ).get(portId, goodId) as { supply: number; demand: number; base_price: number } | undefined;

        // Ports buy anything (even if not in their inventory) at base price - allow selling anywhere
        const basePrice = market?.base_price ?? 60;
        const supply = market?.supply ?? 10;
        const demand = market?.demand ?? 5;
        const delta = demand - supply;
        const multiplier = 1 + delta * 0.02;
        const price = Math.round(Math.max(basePrice * 0.5, Math.min(basePrice * 1.5, basePrice * multiplier)));
        const totalRevenue = price * quantity;

        const updatedCargo: CargoItem[] = cargoItem.quantity === quantity
          ? inv.cargoItems.filter(c => c.goodId !== goodId)
          : inv.cargoItems.map(c => c.goodId === goodId ? { ...c, quantity: c.quantity - quantity } : c);

        const newInventory: PlayerInventory = {
          ...inv,
          gold: inv.gold + totalRevenue,
          cargoItems: updatedCargo,
          currentCargo: inv.currentCargo - quantity,
        };

        db.prepare(
          `UPDATE player_inventory SET gold = ?, cargo_json = ? WHERE id = 1`
        ).run(newInventory.gold, JSON.stringify(newInventory.cargoItems));

        const currentState = loadGameState(db);
        db.prepare(
          `INSERT INTO transaction_history (turn, port_id, good_id, action, quantity, price_per_unit, total_value)
           VALUES (?, ?, ?, 'sell', ?, ?, ?)`
        ).run(currentState?.currentTurn ?? 0, portId, goodId, quantity, price, totalRevenue);

        // Port demand decreases slightly (market now more supplied)
        db.prepare(
          `UPDATE port_inventory SET demand = MAX(0, demand - ?) WHERE port_id = ? AND good_id = ?`
        ).run(Math.floor(quantity / 3), portId, goodId);

        return newInventory;
      })();

      return { success: true, inventory: result };
    }

    export function getTransactionHistory(db: Database.Database, limit: number = 10): TransactionRecord[] {
      return db.prepare(
        `SELECT id, turn, port_id as portId, good_id as goodId, action, quantity, price_per_unit as pricePerUnit, total_value as totalValue
         FROM transaction_history ORDER BY id DESC LIMIT ?`
      ).all(limit) as TransactionRecord[];
    }

    import { useState, useEffect, useCallback } from 'react';
    import type { MarketGood, PlayerInventory, TransactionRecord } from '../../shared/game-types';

    export function useTradeState(portId: string | null) {
      const [marketGoods, setMarketGoods] = useState<MarketGood[]>([]);
      const [inventory, setInventory] = useState<PlayerInventory | null>(null);
      const [transactions, setTransactions] = useState<TransactionRecord[]>([]);
      const [loading, setLoading] = useState(false);
      const [error, setError] = useState<string | null>(null);

      const refresh = useCallback(async () => {
        if (!portId) return;
        setLoading(true);
        try {
          const [goods, inv, txs] = await Promise.all([
            window.api.game.getMarketPrices(portId),
            window.api.game.getPlayerInventory(),
            window.api.game.getTransactionHistory(),
          ]);
          setMarketGoods(goods);
          setInventory(inv);
          setTransactions(txs);
        } catch (e) {
          setError(String(e));
        } finally {
          setLoading(false);
        }
      }, [portId]);

      useEffect(() => { refresh(); }, [refresh]);

      // Subscribe to game:updated events so inventory refreshes after buys/sells
      useEffect(() => {
        const unsub = window.api.subscribe('game:updated', () => refresh());
        return unsub;
      }, [refresh]);

      return { marketGoods, inventory, transactions, loading, error, refresh };
    }

    import React, { useState } from 'react';
    import type { MarketGood, PlayerInventory } from '../../shared/game-types';

    interface TradePanelProps {
      portId: string;
      marketGoods: MarketGood[];
      inventory: PlayerInventory | null;
      onBuy: (goodId: string, qty: number) => Promise<void>;
      onSell: (goodId: string, qty: number) => Promise<void>;
    }

    export default function TradePanel({ portId, marketGoods, inventory, onBuy, onSell }: TradePanelProps) {
      const [quantities, setQuantities] = useState<Record<string, number>>({});
      const [feedback, setFeedback] = useState<string | null>(null);

      const getQty = (goodId: string) => quantities[goodId] ?? 1;
      const setQty = (goodId: string, val: number) =>
        setQuantities(prev => ({ ...prev, [goodId]: Math.max(1, val) }));

      const handleBuy = async (goodId: string) => {
        try {
          await onBuy(goodId, getQty(goodId));
          setFeedback(`Bought ${getQty(goodId)}× ${goodId}`);
          setTimeout(() => setFeedback(null), 2000);
        } catch (e) {
          setFeedback(`Error: ${String(e)}`);
          setTimeout(() => setFeedback(null), 3000);
        }
      };

      const handleSell = async (goodId: string) => {
        try {
          await onSell(goodId, getQty(goodId));
          setFeedback(`Sold ${getQty(goodId)}× ${goodId}`);
          setTimeout(() => setFeedback(null), 2000);
        } catch (e) {
          setFeedback(`Error: ${String(e)}`);
          setTimeout(() => setFeedback(null), 3000);
        }
      };

      const cargoUsed = inventory?.currentCargo ?? 0;
      const cargoMax = inventory?.cargoCapacity ?? 50;
      const gold = inventory?.gold ?? 0;

      return (
        <div className="p-4 flex flex-col gap-3" style={{ fontFamily: 'Lora, serif', fontSize: '0.85rem' }}>
          {/* Header: gold + cargo bar */}
          <div style={{ borderBottom: '1px solid #c4956a', paddingBottom: '0.5rem' }}>
            <div style={{ display: 'flex', justifyContent: 'space-between', fontFamily: 'Cinzel, serif', fontSize: '0.8rem', color: '#3d2b1f' }}>
              <span>⚖ {gold} denarii</span>
              <span>⚓ {cargoUsed}/{cargoMax}</span>
            </div>
            {/* Cargo bar */}
            <div style={{ height: 6, background: '#e8c48a', borderRadius: 3, marginTop: 4 }}>
              <div style={{ height: '100%', width: `${(cargoUsed / cargoMax) * 100}%`, background: '#7a5c45', borderRadius: 3, transition: 'width 0.3s' }} />
            </div>
          </div>

          {feedback && (
            <div style={{ padding: '4px 8px', background: 'rgba(122,92,69,0.15)', border: '1px solid #c4956a', borderRadius: 3, fontSize: '0.78rem', color: '#3d2b1f' }}>
              {feedback}
            </div>
          )}

          {/* Goods list */}
          <div className="flex flex-col gap-2">
            {marketGoods.map(good => {
              const cargoQty = inventory?.cargoItems.find(c => c.goodId === good.goodId)?.quantity ?? 0;
              const qty = getQty(good.goodId);
              const canAfford = gold >= good.currentPrice * qty;
              const hasSpace = cargoUsed + qty <= cargoMax;
              const hasEnoughToSell = cargoQty >= qty;

              return (
                <div key={good.goodId} style={{ border: '1px solid #c4956a', borderRadius: 4, padding: '0.5rem', background: 'rgba(240,217,181,0.3)' }}>
                  <div style={{ display: 'flex', justifyContent: 'space-between', marginBottom: 4 }}>
                    <span style={{ fontWeight: 600, color: '#3d2b1f', textTransform: 'capitalize' }}>{good.name}</span>
                    <span style={{ color: '#7a5c45', fontFamily: 'Cinzel, serif', fontSize: '0.78rem' }}>{good.currentPrice}⚖</span>
                  </div>
                  <div style={{ fontSize: '0.72rem', color: '#b8894a', marginBottom: 4 }}>
                    {good.description} {cargoQty > 0 && <span style={{ color: '#5a3e2b' }}>· You carry: {cargoQty}</span>}
                  </div>
                  <div style={{ display: 'flex', gap: 6, alignItems: 'center' }}>
                    <input
                      type="number"
                      min={1}
                      value={qty}
                      onChange={e => setQty(good.goodId, parseInt(e.target.value) || 1)}
                      style={{ width: 44, padding: '2px 4px', border: '1px solid #c4956a', borderRadius: 3, fontFamily: 'Lora, serif', fontSize: '0.8rem', background: '#fdf5e6' }}
                    />
                    <button
                      onClick={() => handleBuy(good.goodId)}
                      disabled={!canAfford || !hasSpace}
                      style={{ padding: '3px 10px', fontFamily: 'Cinzel, serif', fontSize: '0.7rem', background: canAfford && hasSpace ? '#7a5c45' : '#c4a882', color: '#f0d9b5', border: 'none', borderRadius: 3, cursor: canAfford && hasSpace ? 'pointer' : 'not-allowed' }}
                    >
                      Buy

                      disabled={!hasEnoughToSell}
                      style={{ padding: '3px 10px', fontFamily: 'Cinzel, serif', fontSize: '0.7rem', background: hasEnoughToSell ? '#5a3e2b' : '#c4a882', color: '#f0d9b5', border: 'none', borderRadius: 3, cursor: hasEnoughToSell ? 'pointer' : 'not-allowed' }}
                    >
                      Sell
                    </button>
                  </div>
                </div>
              );
            })}
          </div>
        </div>
      );
    }
