    // ── Event Roll ────────────────────────────────────────────────────────────

    // Weighted random: 25% storm, 20% pirate, 55% calm
    const EVENT_WEIGHTS: Record<string, number> = {
      storm: 25,
      pirate: 20,
      calm: 55,
    };

    export function rollVoyageEvent(turn: number, _fromPortId?: string): VoyageEvent | null {
      // 70% chance of any event per voyage (30% "nothing happens" = skip interrupt)
      if (Math.random() > 0.7) return null;

      // Weighted type selection
      const total = Object.values(EVENT_WEIGHTS).reduce((a, b) => a + b, 0);
      let rand = Math.random() * total;
      let selectedType: string = 'calm';

      for (const [type, weight] of Object.entries(EVENT_WEIGHTS)) {
        rand -= weight;
        if (rand <= 0) { selectedType = type; break; }
      }

      // Random event of that type from database
      const pool = EVENT_DATABASE.filter(e => e.type === selectedType);
      if (pool.length === 0) return null;
      return pool[Math.floor(Math.random() * pool.length)];
    }

    // ── Event Resolution ──────────────────────────────────────────────────────

    export interface EventOutcome {
      goldDelta: number;         // negative = loss, positive = gain
      cargoDelta: Partial<Record<string, number>>; // goodId -> quantity delta (negative = lost)
      turnDelta: number;         // extra turns added to voyage (delay)
      message: string;           // human-readable result to display
    }

    export function resolveEventChoice(
      event: VoyageEvent,
      choiceId: string,
      inventory: PlayerInventory
    ): EventOutcome {
      const choice = event.choices.find(c => c.id === choiceId);
      if (!choice) throw new Error(`Unknown choice ${choiceId} for event ${event.id}`);

      // Parse outcome based on event id + choice id
      // Keep resolution logic deterministic except where RNG is explicitly noted
      return resolveByEventAndChoice(event, choice.id, inventory);
    }

    function resolveByEventAndChoice(event: VoyageEvent, choiceId: string, inventory: PlayerInventory): EventOutcome {
      const cargoTotalValue = inventory.cargoItems.reduce((sum, c) => sum + c.quantity * c.boughtAtPrice, 0);
      const cargoCount = inventory.currentCargo;

      // Helper: lose percent of all cargo (proportional reduction)
      const loseCargoPercent = (pct: number): Partial<Record<string, number>> => {
        const result: Partial<Record<string, number>> = {};
        for (const item of inventory.cargoItems) {
          const loss = Math.floor(item.quantity * pct);
          if (loss > 0) result[item.goodId] = -loss;
        }
        return result;
      };

      // Helper: lose a specific good (all of it)
      const loseAllOfGood = (goodId: string): Partial<Record<string, number>> => {
        const item = inventory.cargoItems.find(c => c.goodId === goodId);
        return item ? { [goodId]: -item.quantity } : {};
      };

      // Helper: lose lowest-value cargo type
      const loseLowestCargo = (): Partial<Record<string, number>> => {
        if (inventory.cargoItems.length === 0) return {};

      const { resolveEventChoice } = await import('../../game/events');
      const inv = getPlayerInventory(db);
      const event = pendingVoyageEvent;
      if (!event || event.id !== eventId) {
        return { success: false, error: 'No pending event or event mismatch' };
      }

      const outcome = resolveEventChoice(event, choiceId, inv);

      // Apply outcome to inventory
      db.transaction(() => {
        // Apply goldDelta
        if (outcome.goldDelta !== 0) {
          db.prepare(`UPDATE player_inventory SET gold = MAX(0, gold + ?) WHERE id = 1`).run(outcome.goldDelta);
        }

        // Apply cargoDelta
        const freshInv = getPlayerInventory(db);
        for (const [goodId, delta] of Object.entries(outcome.cargoDelta)) {
          if (delta === undefined) continue;
          const item = freshInv.cargoItems.find(c => c.goodId === goodId);
          if (delta > 0) {
            // Gain cargo
            const updatedCargo = item
              ? freshInv.cargoItems.map(c => c.goodId === goodId ? { ...c, quantity: c.quantity + delta } : c)
              : [...freshInv.cargoItems, { goodId, quantity: delta, boughtAtPrice: 0, boughtAtPort: 'salvaged' }];
            db.prepare(`UPDATE player_inventory SET cargo_json = ?, cargo_capacity = MAX(cargo_capacity, ?) WHERE id = 1`)
              .run(JSON.stringify(updatedCargo), updatedCargo.reduce((s, c) => s + c.quantity, 0));
          } else if (delta < 0 && item) {
            // Lose cargo
            const newQty = Math.max(0, item.quantity + delta);
            const updatedCargo = newQty === 0
              ? freshInv.cargoItems.filter(c => c.goodId !== goodId)
              : freshInv.cargoItems.map(c => c.goodId === goodId ? { ...c, quantity: newQty } : c);
            db.prepare(`UPDATE player_inventory SET cargo_json = ? WHERE id = 1`).run(JSON.stringify(updatedCargo));
          }
        }
      })();

      // Recalculate currentCargo
      const finalInv = getPlayerInventory(db);
      db.prepare(`UPDATE player_inventory SET cargo_json = ? WHERE id = 1`).run(
        JSON.stringify(finalInv.cargoItems)
      );

      // Transition FSM: encountered → idle (RESOLVE action)
      const current = loadGameState(db) ?? { type: 'idle', currentTurn: 0 };
      const nextState = performAction(current as any, { type: 'RESOLVE' });
      saveGameStateAtomic(db, nextState as any);

      pendingVoyageEvent = null;

      // Broadcast updated state
      const { BrowserWindow } = await import('electron');
      for (const win of BrowserWindow.getAllWindows()) {
        if (!win.isDestroyed()) win.webContents.send('subscribe:game:updated', nextState);
      }

      return { success: true, outcome, message: outcome.message };
    });
