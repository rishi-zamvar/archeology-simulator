    ```typescript
    export interface CachedArtifact {
      id: string;
      goodId: string;
      museumId: string;
      imageUrl: string;
      title: string;
      description: string;
      cacheTimestamp: string;
    }

    export function getArtifactsForGood(db: Database.Database, goodId: string): CachedArtifact[] {
      return db.prepare(`
        SELECT id, good_id as goodId, museum_id as museumId, image_url as imageUrl,
               title, description, cache_timestamp as cacheTimestamp
        FROM artifact_cache
        WHERE good_id = ? AND datetime(cache_timestamp) > datetime('now', '-24 hours')
        LIMIT 3
      `).all(goodId) as CachedArtifact[];
    }

    export function cacheArtifact(db: Database.Database, artifact: CachedArtifact): void {
      db.prepare(`
        INSERT OR REPLACE INTO artifact_cache
          (id, good_id, museum_id, image_url, title, description, cache_timestamp)
        VALUES (?, ?, ?, ?, ?, ?, ?)
      `).run(
        artifact.id, artifact.goodId, artifact.museumId, artifact.imageUrl,
        artifact.title, artifact.description, artifact.cacheTimestamp
      );
    }
    ```

    **src/renderer/hooks/useArtifactImages.ts** — Create hook:
    ```typescript
    import { useEffect, useState } from 'react';
    import type { CachedArtifact } from '../../shared/ipc-types'; // or wherever CachedArtifact is exported

    export function useArtifactImages(goodIds: string[]) {
      const [artifacts, setArtifacts] = useState<Map<string, CachedArtifact[]>>(new Map());

      const key = goodIds.slice().sort().join(',');

      useEffect(() => {
        if (goodIds.length === 0) return;
        let cancelled = false;

        Promise.all(goodIds.map((id) => window.api.game.getArtifacts(id)))
          .then((results) => {
            if (cancelled) return;
            const map = new Map<string, CachedArtifact[]>();
            goodIds.forEach((id, idx) => map.set(id, results[idx] || []));
            setArtifacts(map);
          })
          .catch(console.error);

        return () => { cancelled = true; };
      }, [key]);

      return artifacts;
    }
    ```
