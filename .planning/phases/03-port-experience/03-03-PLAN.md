    **src/renderer/hooks/useAudioManager.ts** — Create AudioManager singleton hook:
    ```typescript
    // Singleton pattern — AudioContext must persist across renders
    const AudioManager = {
      context: null as AudioContext | null,
      currentSource: null as AudioBufferSourceNode | null,
      currentGain: null as GainNode | null,

      async playAmbient(url: string, volume = 0.4, fadeInMs = 2000) {
        try {
          if (!this.context) {
            this.context = new (window.AudioContext || (window as any).webkitAudioContext)();
          }
          await this.context.resume(); // Handle suspended context
          const response = await fetch(url);
          if (!response.ok) return; // Graceful fail if asset missing
          const arrayBuffer = await response.arrayBuffer();
          const buffer = await this.context.decodeAudioData(arrayBuffer);
          if (this.currentSource) {
            try { this.currentSource.stop(); } catch {}
          }
          const source = this.context.createBufferSource();
          source.buffer = buffer;
          source.loop = true;
          const gain = this.context.createGain();
          gain.gain.value = 0;
          gain.gain.linearRampToValueAtTime(volume, this.context.currentTime + fadeInMs / 1000);
          source.connect(gain);
          gain.connect(this.context.destination);
          source.start(0);
          this.currentSource = source;
          this.currentGain = gain;
        } catch (err) {
          console.warn('Audio playback failed (non-fatal):', err);
        }
      },

      stopAmbient(fadeOutMs = 1000) {
        if (!this.context || !this.currentGain || !this.currentSource) return;
        const stopAt = this.context.currentTime + fadeOutMs / 1000;
        this.currentGain.gain.linearRampToValueAtTime(0, stopAt);
        const src = this.currentSource;
        setTimeout(() => { try { src.stop(); } catch {} }, fadeOutMs + 50);
        this.currentSource = null;
        this.currentGain = null;
      },
    };

    export function useAudioManager() {
      return AudioManager;
    }
    ```
