    Create src/renderer/stores/uiStore.ts using Zustand:
    ```typescript
    interface UIState {
      leftPanelOpen: boolean;
      rightPanelOpen: boolean;
      activePanel: 'inventory' | 'port' | 'log' | null;
      toggleLeftPanel: () => void;
      toggleRightPanel: () => void;
      setActivePanel: (panel: UIState['activePanel']) => void;
    }
    export const useUIStore = create<UIState>((set) => ({
      leftPanelOpen: false,
      rightPanelOpen: false,
      activePanel: null,
      toggleLeftPanel: () => set((s) => ({ leftPanelOpen: !s.leftPanelOpen })),
      toggleRightPanel: () => set((s) => ({ rightPanelOpen: !s.rightPanelOpen })),
      setActivePanel: (panel) => set({ activePanel: panel }),
    }));
    ```

    Create src/renderer/hooks/useGameState.ts following RESEARCH.md Pattern 4 exactly:
    - useEffect #1: window.api.game.load() on mount, set state, set loading false
    - useEffect #2: const cleanup = window.api.subscribe('game:updated', callback); return () => cleanup()
    - CRITICAL: The cleanup return in useEffect #2 prevents memory leaks — do not omit it
    - Returns { state: GameState | null, isLoading: boolean, error: Error | null, updateGame: (action) => Promise<GameResult> }
