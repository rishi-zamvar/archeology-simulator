  <action>
    Create .github/workflows/ci.yml with three build jobs running in parallel:

    ```yaml
    name: CI
    on:
      push:
        branches: [main]
      pull_request:
        branches: [main]

    jobs:
      test:
        runs-on: ubuntu-latest
        steps:
          - uses: actions/checkout@v4
          - uses: actions/setup-node@v4
            with: { node-version: '20', cache: 'npm' }
          - run: npm ci
          - run: npx tsc --noEmit
          - run: npx vitest run

      build-macos:
        runs-on: macos-latest
        needs: test
        steps:
          - uses: actions/checkout@v4
          - uses: actions/setup-node@v4
            with: { node-version: '20', cache: 'npm' }
          - run: npm ci
          - run: npm run build
          env:
            ELECTRON_BUILDER_ALLOW_UNRESOLVED_DEPENDENCIES: true

      build-windows:
        runs-on: windows-latest
        needs: test
        steps:
          - uses: actions/checkout@v4
          - uses: actions/setup-node@v4
            with: { node-version: '20', cache: 'npm' }
          - run: npm ci
          - run: npm run build
          env:
            ELECTRON_BUILDER_ALLOW_UNRESOLVED_DEPENDENCIES: true

      build-linux:
        runs-on: ubuntu-latest
        needs: test
        steps:
          - uses: actions/checkout@v4
          - uses: actions/setup-node@v4
            with: { node-version: '20', cache: 'npm' }
          - run: npm ci
          - run: npm run build
          env:
            ELECTRON_BUILDER_ALLOW_UNRESOLVED_DEPENDENCIES: true
    ```

    Key: test job runs first (tsc + vitest), then 3 platform builds run in parallel after test passes. This validates better-sqlite3 native compilation on all three platforms — the Phase 1 blocker identified in STATE.md.
