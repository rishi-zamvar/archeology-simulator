**Check 2: Every image file on disk has a manifest entry**
```bash
python3 -c "
import json, os, glob

with open('resources/data/curated-images.json') as f:
    data = json.load(f)

manifest_files = {e['filePath'] for e in data}

# Check port images
orphans = []
for f in glob.glob('resources/images/ports/*.jpg') + glob.glob('resources/images/ports/*.png'):
    rel = f.replace('resources/', '')
    if rel not in manifest_files:
        orphans.append(f)

# Check trade good images
for good in ['wheat','olive_oil','wine','textiles','spices','bronze','timber','pottery']:
    for f in glob.glob(f'resources/images/{good}/{good}_*.jpg') + glob.glob(f'resources/images/{good}/{good}_*.png'):
        rel = f.replace('resources/', '')
        if rel not in manifest_files:
            orphans.append(f)

if orphans:
    print('ORPHAN FILES (no manifest entry):')
    for o in orphans: print(f'  {o}')
else:
    print('OK: No orphan image files')
"
```

```bash
cat resources/data/curated-images.json | python3 -c "
import json, sys
data = json.load(sys.stdin)
valid_ports = {'athens','corinth','sparta','miletos','ephesos','byzantion','rhodes','knossos',
               'syracuse','massalia','taras','ostia','carthage','utica','leptis',
               'alexandria','naukratis','memphis','tyre','sidon','byblos','kition','salamis_cy',
               'emporion','gadir'}
valid_goods = {'wheat','olive_oil','wine','textiles','spices','bronze','timber','pottery'}

port_entries = [e for e in data if 'portId' in e]
good_entries = [e for e in data if 'goodId' in e]

# Check ports
found_ports = {e['portId'] for e in port_entries}
bad_ports = found_ports - valid_ports
missing_ports = valid_ports - found_ports
if bad_ports: print(f'INVALID portIds: {bad_ports}')
if missing_ports: print(f'MISSING ports: {missing_ports}')

# Check goods
found_goods = {e['goodId'] for e in good_entries}
bad_goods = found_goods - valid_goods
missing_goods = valid_goods - found_goods
if bad_goods: print(f'INVALID goodIds: {bad_goods}')
if missing_goods: print(f'MISSING goods: {missing_goods}')

# Check required fields
for e in data:
    for k in ('id', 'filePath', 'attribution', 'sourceUrl'):
        if k not in e or not e[k]:
            print(f'MISSING/EMPTY field {k} in entry {e.get(\"id\",\"?\")}: {e}')

# Check unique IDs
ids = [e['id'] for e in data]
dupes = set(x for x in ids if ids.count(x) > 1)
if dupes: print(f'DUPLICATE IDs: {dupes}')

if not bad_ports and not missing_ports and not bad_goods and not missing_goods and not dupes:
    print(f'OK: {len(port_entries)} port entries, {len(good_entries)} good entries, all IDs valid, no duplicates')
else:
    sys.exit(1)
"
```
