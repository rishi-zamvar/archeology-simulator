```typescript
export function seedCuratedImages(db: Database.Database): void {
  // eslint-disable-next-line @typescript-eslint/no-require-imports
  const manifest = require('../../../resources/data/curated-images.json') as Array<{
    id: string;
    goodId?: string;
    portId?: string;
    filePath: string;
    attribution: string;
    sourceUrl?: string;
    createdAt?: string;
  }>;

  const insert = db.prepare(`
    INSERT OR IGNORE INTO curated_images
      (id, good_id, port_id, file_path, attribution, source_url, created_at)
    VALUES (?, ?, ?, ?, ?, ?, ?)
  `);

  for (const image of manifest) {
    insert.run(
      image.id,
      image.goodId ?? null,
      image.portId ?? null,
      image.filePath,
      image.attribution,
      image.sourceUrl ?? null,
      image.createdAt ?? new Date().toISOString()
    );
  }
}
```

**3. getCuratedImagesForGood and getCuratedImagesForPort functions**:
```typescript
export function getCuratedImagesForGood(db: Database.Database, goodId: string): CuratedImage[] {
  return db.prepare(`
    SELECT id,
           good_id    AS goodId,
           port_id    AS portId,
           file_path  AS filePath,
           attribution,
           source_url AS sourceUrl,
           created_at AS createdAt
    FROM curated_images
    WHERE good_id = ?
    LIMIT 5
  `).all(goodId) as CuratedImage[];
}

export function getCuratedImagesForPort(db: Database.Database, portId: string): CuratedImage[] {
  return db.prepare(`
    SELECT id,
           good_id    AS goodId,
           port_id    AS portId,
           file_path  AS filePath,
           attribution,
           source_url AS sourceUrl,
           created_at AS createdAt
    FROM curated_images
    WHERE port_id = ?
    LIMIT 3
  `).all(portId) as CuratedImage[];
}
```

No new imports needed at the top of queries.ts — `Database` is already imported and `require` is available in the Node.js/Electron main process.
