At the end of the `registerGameIPCHandlers` function body (before the closing `}`), add the new handler:

```typescript
  /**
   * game:get-curated-images — retrieve offline-available curated image metadata.
   *
   * Query: { goodId?: string; portId?: string }
   * Returns CuratedImage[] with filePath resolved to absolute path.
   * fileExists field lets renderer skip images not yet bundled (Phase 5 populates them).
   *
   * Path resolution:
   *   development: __dirname points to src/main — go up two levels to reach project root's resources/
   *   production: app.getAppPath() points to app.asar root where resources/ is bundled
   */
  ipcMain.handle('game:get-curated-images', async (_event, query: { goodId?: string; portId?: string }) => {
    let images;

    if (query.goodId) {
      images = getCuratedImagesForGood(db, query.goodId);
    } else if (query.portId) {
      images = getCuratedImagesForPort(db, query.portId);
    } else {
      return [];
    }

    const resourcesBase = process.env['NODE_ENV'] === 'development'
      ? path.join(__dirname, '../../resources')
      : path.join(app.getAppPath(), 'resources');

    return images.map(image => ({
      ...image,
      filePath: path.join(resourcesBase, image.filePath),
      fileExists: fs.existsSync(path.join(resourcesBase, image.filePath)),
    }));
  });
```
