```typescript
// --- v2.0 Dialogue Engine Types ---

export type ConditionType = 'quest' | 'reputation' | 'visited_port' | 'has_cargo' | 'turn_range' | 'season';

export interface QuestCondition { type: 'quest'; id: string; status: QuestStatus | 'not_started'; }
export interface ReputationCondition { type: 'reputation'; region: string; min?: number; max?: number; }
export interface VisitedPortCondition { type: 'visited_port'; portId: string; }
export interface HasCargoCondition { type: 'has_cargo'; goodId: string; minQuantity?: number; }
export interface TurnRangeCondition { type: 'turn_range'; min?: number; max?: number; }
export interface SeasonCondition { type: 'season'; season: string; }

export type DialogueCondition =
  | QuestCondition | ReputationCondition | VisitedPortCondition
  | HasCargoCondition | TurnRangeCondition | SeasonCondition;

export type CompositeCondition =
  | DialogueCondition
  | { $and: CompositeCondition[] }
  | { $or: CompositeCondition[] }
  | { $not: CompositeCondition };

export type EffectType = 'quest_update' | 'reputation_change' | 'port_unlock' | 'journal_entry' | 'trade_offer';

export interface DialogueEffect {
  type: EffectType;
  [key: string]: unknown;
}

export interface DialogueChoice {
  text: string;
  next: string;
  effects: DialogueEffect[];
  conditions?: CompositeCondition;
}

export interface DialogueNode {
  id: string;
  speaker: string;
  text: string;
  conditions?: CompositeCondition;
  choices: DialogueChoice[];
}

export interface DialogueFile {
  npc_id: string;
  port_id: string;
  root_node: string;
  nodes: DialogueNode[];
}
```

2. Create src/game/dialogue.ts as a pure module (~200 lines, zero side effects, zero imports from Electron or Node.js):

- Define `GameContext` interface: `{ questProgress: Map<string, QuestStatus>; reputation: Map<string, number>; visitedPorts: Set<string>; cargo: Array<{ goodId: string; quantity: number }>; currentTurn: number; season: string; }`

- Export `evaluateCondition(condition: CompositeCondition, ctx: GameContext): boolean`:
  - For `$and`: every sub-condition must be true
  - For `$or`: at least one sub-condition true
  - For `$not`: sub-condition must be false
  - For leaf conditions, dispatch on `type`:
    - `quest`: check `ctx.questProgress.get(id) === status` (treat missing as 'not_started')
    - `reputation`: check `ctx.reputation.get(region)` against min/max bounds
    - `visited_port`: check `ctx.visitedPorts.has(portId)`
    - `has_cargo`: check cargo array for goodId with quantity >= minQuantity (default 1)
    - `turn_range`: check currentTurn against min/max bounds
    - `season`: check `ctx.season === season`
  - Use exhaustive switch on `type` with a default that returns false

- Export `getAvailableChoices(node: DialogueNode, ctx: GameContext): DialogueChoice[]`:
  - Filter node.choices: include choice only if choice.conditions is undefined OR evaluateCondition returns true
  - Per locked decision: failing conditions means HIDDEN, not greyed out

- Export `getNextNode(nodes: DialogueNode[], nodeId: string): DialogueNode | undefined`:
  - Find node by id in array

- Export `findEntryNode(file: DialogueFile, ctx: GameContext): DialogueNode | undefined`:
  - Start from root_node. If root node has conditions that fail, return undefined.
  - Otherwise return the root node.

Do NOT import fs, path, ipcMain, or any Node/Electron API. This module is pure.
