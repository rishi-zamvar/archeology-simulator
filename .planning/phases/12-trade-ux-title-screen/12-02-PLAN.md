```typescript
// game:hasSave -- check if a save exists without loading the full state
wrapHandler<boolean>(
  'game:hasSave',
  async () => {
    const row = db.prepare('SELECT 1 FROM game_state WHERE id = 1').get();
    return row !== undefined;
  },
);

// game:newGame -- reset all game data and return fresh initial state
wrapHandler<GameState>(
  'game:newGame',
  async () => {
    const reset = db.transaction(() => {
      db.prepare('DELETE FROM game_state').run();
      db.prepare('DELETE FROM transaction_history').run();
      db.prepare('DELETE FROM journal_entries').run();
      db.prepare("DELETE FROM player_inventory").run();
      // Re-insert fresh state
      saveGameStateAtomic(db, INITIAL_STATE);
    });
    reset();
    return INITIAL_STATE;
  },
);
```

Note: `wrapHandler` is already imported in save.ts (from '../ipc-utils'). `INITIAL_STATE` and `saveGameStateAtomic` are already in scope.

```typescript
const [screen, setScreen] = useState<'title' | 'game'>('title');
const [hasSave, setHasSave] = useState(false);

// Check for existing save on mount (lightweight IPC, no game state loading)
useEffect(() => {

    if (res.success) setHasSave(res.data);
```

Add handlers:

```typescript
const handleContinue = useCallback(() => {

  }).catch((err: unknown) => showError('Failed to start new game: ' + (err instanceof Error ? err.message : 'Unknown error')));
}, [showError]);
```

Gate the render -- at the start of the return statement, before the existing game UI:
