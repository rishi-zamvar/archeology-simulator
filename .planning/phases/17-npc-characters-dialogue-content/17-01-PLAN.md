    ```typescript
    npcDialogueState: Map<string, { timeSpoken: number; lastNodeId: string | null }>;
    ```

    Then in buildGameContext (src/main/ipc/dialogue.ts), populate it from the npc_dialogue_state table:
    ```typescript
    const npcState = new Map();
    const npcRows = db.prepare('SELECT npc_id, times_spoken, last_node_id FROM npc_dialogue_state').all();
    for (const row of npcRows) {
      npcState.set(row.npc_id, { timeSpoken: row.times_spoken, lastNodeId: row.last_node_id });
    }
    ```

    case 'talk_to_npc': {

    }
